============================================================
🦌 DEER CAMOUFLAGE DETECTION - QUICK EXECUTION GUIDE
============================================================

------------------------------------------------------------
TERMINAL 1: START SIMULATION (Gazebo World + Robot)
------------------------------------------------------------
cd ~/Desktop/animal_detection
./start_docker.sh

# Inside Docker container:
source /opt/ros/humble/setup.bash
source install/setup.bash
export GAZEBO_MODEL_PATH=/workspace/models:$GAZEBO_MODEL_PATH

# Option A: 3D Gazebo GUI Mode
ros2 launch animal_gazebo simulation.launch.py gui:=true

# Option B: Headless Mode (Fast, Low CPU/RAM)
ros2 launch animal_gazebo simulation.launch.py gui:=false

------------------------------------------------------------
TERMINAL 2: RUN CAMOUFLAGE YOLO DETECTION NODE
------------------------------------------------------------
docker exec -it $(docker ps -q -f ancestor=animal_detection_ros2:humble | head -1) bash
source /opt/ros/humble/setup.bash
source install/setup.bash

# Run YOLOv8 Camouflage Recognition Node (with CLAHE texture enhancement)
ros2 launch animal_recognition recognition.launch.py

------------------------------------------------------------
TERMINAL 3: VIEW LIVE ANNOTATED HUD CAMERA FEED
------------------------------------------------------------
docker exec -it $(docker ps -q -f ancestor=animal_detection_ros2:humble | head -1) bash
source /opt/ros/humble/setup.bash

# Launch RQT Image View to see raw or annotated detections:
# Select topic: /animal_detection/annotated_image
ros2 run rqt_image_view rqt_image_view

------------------------------------------------------------
TERMINAL 4: ROBOT KEYBOARD TELEOPERATION (Drive around)
------------------------------------------------------------
docker exec -it $(docker ps -q -f ancestor=animal_detection_ros2:humble | head -1) bash
source /opt/ros/humble/setup.bash

# Drive robot with keyboard (i, j, k, l, etc.)
ros2 run teleop_twist_keyboard teleop_twist_keyboard --ros-args -r cmd_vel:=/animal_robot/cmd_vel

------------------------------------------------------------
OPTIONAL: FINE-TUNING & DATASET COLLECTION
------------------------------------------------------------
# 1. Collect dataset frames while driving around camouflaged targets:
ros2 run animal_recognition /workspace/training/capture_dataset.py

# 2. Fine-tune YOLOv8 on the captured dataset:
python3 /workspace/training/train_camouflage_yolo.py --epochs 30 --batch 8
